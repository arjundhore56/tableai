import OpenAI from "openai"; const client = new OpenAI({ apiKey: process.env.OPENAI_API_KEY, }); export default async function handler(req, res) { // Only allow POST requests if (req.method !== "POST") { return res.status(405).json({ error: "Method not allowed", }); } // Check API key if (!process.env.OPENAI_API_KEY) { return res.status(500).json({ error: "OPENAI_API_KEY is missing in Vercel.", }); } try { const { tool, business, request } = req.body || {}; // Check required information if (!business) { return res.status(400).json({ error: "Business profile is missing.", }); } if (!tool) { return res.status(400).json({ error: "TABLEAI tool is missing.", }); } const prompt = ` You are TABLEAI, an AI marketing and growth assistant for businesses and startups. Your job is to create practical, natural and ready-to-use marketing output. IMPORTANT RULES: - Never invent prices, discounts, products, services, reviews, awards, statistics or business claims. - Only use information provided in the business profile. - If something is missing, make it clear that it is a suggestion. - Keep the writing natural and human. - Avoid generic AI-sounding language. - Avoid excessive emojis. - Make the result useful for a busy business owner. - Give specific and actionable ideas. - Write in a clear and easy-to-use format. BUSINESS PROFILE: ${JSON.stringify(business, null, 2)} TABLEAI TOOL: ${tool} USER REQUEST: ${request || "Create the most useful output for this request."} Create the final answer for the business owner now. `; const response = await client.responses.create({ model: "gpt-5.6-luna", input: prompt, }); return res.status(200).json({ success: true, output: response.output_text, }); } catch (error) { console.error("TABLEAI ERROR:", error); return res.status(500).json({ success: false, error: error?.message || "Unknown TABLEAI server error.", }); } }
